    f_mid_m = (v0_mps^2/g_mps2)*sind(2*mid_deg) - target_m;
    if abs(f_mid_m) < tolerance_m
        break
    end
    f_bis_left_m = (v0_mps^2/g_mps2)*sind(2*bis_left_deg) - target_m;
    if f_bis_left_m*f_mid_m < 0
